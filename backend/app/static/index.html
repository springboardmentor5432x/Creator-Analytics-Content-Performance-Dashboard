SQL DDL Definition Code

-- SQL Schema DDL Script for Module 1
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    hashed_password VARCHAR(255) NOT NULL,
    role VARCHAR(30) NOT NULL CHECK (role IN ('Creator', 'Agency', 'Marketing Team', 'Administrator')),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE creator_profiles (
    id SERIAL PRIMARY KEY,
    user_id INT UNIQUE REFERENCES users(id) ON DELETE CASCADE,
    handle VARCHAR(50),
    niche VARCHAR(100) DEFAULT 'Tech & Content',
    youtube_channel VARCHAR(100),
    total_followers INT DEFAULT 0
);
